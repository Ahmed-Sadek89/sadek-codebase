// To deploy a NestJS project on Vercel, you'll typically use Vercel's custom server support. 
// The process generally involves creating a `vercel.json`, an `api/index.ts` (for entrypoint), and adjusting your NestJS main file. 
// Below are the essential steps and example files:

// 1. Create a vercel.json file at your root:

/*
{
  "version": 2,
  "builds": [
    { "src": "api/index.ts", "use": "@vercel/node" }
  ],
  "routes": [
    { "src": "/(.*)", "dest": "/api/index.ts" }
  ]
}
*/

// 2. In your NestJS project, create an `api/index.ts` file. This will serve as the entry point for Vercel's serverless function.

// api/index.ts
import { createServer, proxy } from 'vercel-node-serverless';
import { NestFactory } from '@nestjs/core';
import { AppModule } from '../src/app.module';

const server = createServer();

async function bootstrap() {
    const app = await NestFactory.create(AppModule, server);
    await app.init();
}

bootstrap();

export default (req, res) => {
    server.emit('request', req, res);
};

// 3. Make sure to update your tsconfig.json to include api directory if not already.
// 4. Remove or ignore default NestJS build scripts that serve on their own port - Vercel will invoke the handler.

// 5. Optional: update package.json scripts for local development:

/*
"scripts": {
  "start:dev": "nest start --watch",
  "build": "nest build"
}
*/

// 6. Then, simply run `vercel` in your root directory and follow the prompts to deploy.
